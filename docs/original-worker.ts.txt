/** Cloudflare Worker entry point for the vinext-starter template. */
import { handleImageOptimization, DEFAULT_DEVICE_SIZES, DEFAULT_IMAGE_SIZES } from "vinext/server/image-optimization";
import handler from "vinext/server/app-router-entry";
import { currentData, currentFile, dataAdmin } from "./dashboard-data";
import { reportEndpoint, reportGrant, snapshotFile } from "./report-service";
import dashboardHtmlSource from "../private/dashboard-source.html?raw";
import permitsDashboardHtmlSource from "../private/permits-dashboard.html?raw";

interface Env {
  ASSETS: Fetcher;
  SAL_PORTAL_PASSWORD?: string;
  SAL_ADMIN_USERNAME?: string;
  SAL_ADMIN_PASSWORD?: string;
  DB: D1Database;
  REPORT_FILES: R2Bucket;
  REPORT_SYNC_SECRET?: string;
  REPORT_EMAIL_HOOK?: string;
  REPORT_EMAIL_ENABLED?: string;
  IMAGES: {
    input(stream: ReadableStream): {
      transform(options: Record<string, unknown>): {
        output(options: { format: string; quality: number }): Promise<{ response(): Response }>;
      };
    };
  };
}

type Employee = {
  id: number; name: string; username: string; password_hash: string; password_salt: string;
  can_amos: number; can_aog: number; can_fak: number; can_marshaling: number; can_permits: number;
  can_export: number; is_admin: number; active: number; created_at: string; updated_at: string;
};

const encoder = new TextEncoder();
const hex = (buffer: ArrayBuffer) => Array.from(new Uint8Array(buffer), b => b.toString(16).padStart(2, "0")).join("");
const sha256 = async (value: string) => hex(await crypto.subtle.digest("SHA-256", encoder.encode(value)));
const randomHex = (length = 16) => { const bytes = new Uint8Array(length); crypto.getRandomValues(bytes); return Array.from(bytes, b => b.toString(16).padStart(2, "0")).join(""); };
async function passwordHash(password: string, salt: string) {
  const material = await crypto.subtle.importKey("raw", encoder.encode(password), "PBKDF2", false, ["deriveBits"]);
  return hex(await crypto.subtle.deriveBits({ name: "PBKDF2", hash: "SHA-256", salt: encoder.encode(salt), iterations: 100000 }, material, 256));
}
const getCookie = (request: Request, name: string) => (request.headers.get("cookie") || "").split(";").map(v => v.trim()).find(v => v.startsWith(name + "="))?.slice(name.length + 1) || "";
const cookie = (name: string, value: string, maxAge = 28800) => `${name}=${value}; Path=/; HttpOnly; Secure; SameSite=Lax; Max-Age=${maxAge}`;
const permissions = (employee: Employee) => ({ amos: !!employee.can_amos, aog: !!employee.can_aog, fak: !!employee.can_fak, marshaling: !!employee.can_marshaling, permits: !!employee.can_permits });
const dashboardKeys = ["amos", "aog", "fak", "marshaling", "permits"] as const;

async function employeeFromRequest(request: Request, env: Env): Promise<Employee | null> {
  const token = getCookie(request, "sal_employee");
  const [encodedUsername, expiryText, signature] = token.split(".");
  let username = "";
  try { const value = encodedUsername.replace(/-/g, "+").replace(/_/g, "/"); username = encodedUsername ? atob(value + "=".repeat((4 - value.length % 4) % 4)) : ""; } catch { return null; }
  const expiry = Number(expiryText);
  if (!username || !expiry || !signature || expiry < Date.now()) return null;
  const employee = await env.DB.prepare("SELECT * FROM employees WHERE username = ? AND active = 1").bind(username).first<Employee>();
  if (!employee) return null;
  const expected = await sha256(`employee-session:${username}:${expiry}:${employee.password_hash}`);
  return expected === signature ? employee : null;
}

function sameOrigin(request: Request) {
  const origin = request.headers.get("origin");
  return !origin || origin === new URL(request.url).origin;
}

interface ExecutionContext {
  waitUntil(promise: Promise<unknown>): void;
  passThroughOnException(): void;
}

// Image security config. SVG sources with .svg extension auto-skip the
// optimization endpoint on the client side (served directly, no proxy).
// To route SVGs through the optimizer (with security headers), set
// dangerouslyAllowSVG: true in next.config.js and uncomment below:
// const imageConfig: ImageConfig = { dangerouslyAllowSVG: true };

const worker = {
  async fetch(request: Request, env: Env, ctx: ExecutionContext): Promise<Response> {
    const url = new URL(request.url);

    const password = env.SAL_PORTAL_PASSWORD || "";
    const expectedSession = password ? await sha256(`sal-session:${password}`) : "";
    const session = getCookie(request, "sal_session");
    const generalAuthorized = Boolean(expectedSession && session === expectedSession);
    const adminUsername = env.SAL_ADMIN_USERNAME || "";
    const adminPassword = env.SAL_ADMIN_PASSWORD || "";
    const expectedAdminSession = adminUsername && adminPassword ? await sha256(`sal-admin:${adminUsername}:${adminPassword}`) : "";
    const employee = await employeeFromRequest(request, env);
    const adminAuthorized = Boolean(expectedAdminSession && getCookie(request, "sal_admin") === expectedAdminSession) || employee?.is_admin === 1;
    const authorized = generalAuthorized || adminAuthorized || Boolean(employee);
    const userPermissions = employee && !adminAuthorized ? permissions(employee) : { amos: true, aog: true, fak: true, marshaling: true, permits: adminAuthorized };

    const reportResponse = await reportEndpoint(request, env, adminAuthorized);
    if (reportResponse) return reportResponse;
    const grant = await reportGrant(request, env);
    if (url.pathname === "/report-render" && !grant) return new Response("Unauthorized", { status: 401 });

    // Keep live dashboard downloads on the Site's own origin. This avoids
    // browser/CORS differences between the embedded dashboard and the sync
    // Worker, while preserving the existing authenticated dashboard access.
    const syncMatch = url.pathname.match(/^\/api\/dashboard-sync\/(amos|aog|fak|marshaling)\/(metadata|file)$/);
    if (syncMatch && request.method === "GET") {
      if (grant && grant.dashboard === syncMatch[1]) {
        if (syncMatch[2] === "metadata") return Response.json({ updatedAt: grant.job.updated_at, dashboard: grant.dashboard, reportSnapshot: true }, { headers: { "Cache-Control": "no-store" } });
        const file = await snapshotFile(env, grant.job, grant.dashboard);
        return file ? new Response(file.body, { headers: { "Content-Type": "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet", "Cache-Control": "no-store" } }) : new Response("Snapshot missing", { status: 404 });
      }
      if (!authorized) return Response.json({ error: "Unauthorized" }, { status: 401, headers: { "Cache-Control": "no-store" } });
      const [, dashboard, resource] = syncMatch;
      try {
        const upstream = resource === "file" ? await currentFile(env, dashboard) : Response.json({ success: true, dashboard, metadata: (await currentData(env, dashboard)).metadata });
        const headers = new Headers();
        headers.set("Cache-Control", "no-store");
        headers.set("Content-Type", upstream.headers.get("Content-Type") || (resource === "file"
          ? "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
          : "application/json; charset=utf-8"));
        return new Response(upstream.body, { status: upstream.status, headers });
      } catch (error) {
        return Response.json(
          { error: "Dashboard sync service unavailable", detail: error instanceof Error ? error.message : String(error) },
          { status: 502, headers: { "Cache-Control": "no-store" } },
        );
      }
    }

    if (url.pathname === "/api/dashboard-sync/client-event" && request.method === "POST") {
      if (!authorized || !sameOrigin(request)) return Response.json({ error: "Unauthorized" }, { status: 401 });
      try {
        const event = await request.json<Record<string, unknown>>();
        console.log("[dashboard-sync]", JSON.stringify({
          dashboard: String(event.dashboard || "").slice(0, 12),
          status: String(event.status || "").slice(0, 24),
          sheet: String(event.sheet || "").slice(0, 80),
          rows: Number(event.rows || 0),
          updatedAt: String(event.updatedAt || "").slice(0, 40),
          message: String(event.message || "").slice(0, 240),
        }));
      } catch {
        console.log("[dashboard-sync] invalid client event");
      }
      return new Response(null, { status: 204, headers: { "Cache-Control": "no-store" } });
    }

    if (url.pathname === "/api/login" && request.method === "POST") {
      const form = await request.formData();
      const submitted = String(form.get("password") || "");
      if (password && submitted === password) {
        return new Response(null, { status: 303, headers: {
          Location: "/portal?welcome=1",
          "Set-Cookie": `sal_session=${expectedSession}; Path=/; HttpOnly; Secure; SameSite=Lax; Max-Age=28800`,
          "Cache-Control": "no-store",
        }});
      }
      return Response.redirect(new URL("/login?error=1", request.url), 303);
    }

    if (url.pathname === "/api/employee-login" && request.method === "POST") {
      const form = await request.formData();
      const username = String(form.get("username") || "").trim().toLowerCase();
      const submitted = String(form.get("password") || "");
      const found = username ? await env.DB.prepare("SELECT * FROM employees WHERE username = ? AND active = 1").bind(username).first<Employee>() : null;
      if (found && await passwordHash(submitted, found.password_salt) === found.password_hash) {
        const expiry = Date.now() + 8 * 60 * 60 * 1000;
        const signature = await sha256(`employee-session:${found.username}:${expiry}:${found.password_hash}`);
        const encodedUsername = btoa(found.username).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
        return new Response(null, { status: 303, headers: { Location: "/portal?welcome=1", "Set-Cookie": cookie("sal_employee", `${encodedUsername}.${expiry}.${signature}`), "Cache-Control": "no-store" } });
      }
      return Response.redirect(new URL("/login?employee_error=1", request.url), 303);
    }

    if (url.pathname === "/api/admin-login" && request.method === "POST") {
      const form = await request.formData();
      const username = String(form.get("username") || "").trim();
      const submitted = String(form.get("password") || "");
      if (adminUsername && adminPassword && username === adminUsername && submitted === adminPassword) {
        return new Response(null, { status: 303, headers: { Location: "/portal?welcome=1", "Set-Cookie": cookie("sal_admin", expectedAdminSession), "Cache-Control": "no-store" } });
      }
      return Response.redirect(new URL("/login?admin_error=1", request.url), 303);
    }

    if (url.pathname === "/api/logout" && request.method === "POST") {
      const headers = new Headers({ Location: "/", "Cache-Control": "no-store" });
      headers.append("Set-Cookie", cookie("sal_session", "", 0));
      headers.append("Set-Cookie", cookie("sal_employee", "", 0));
      headers.append("Set-Cookie", cookie("sal_admin", "", 0));
      return new Response(null, { status: 303, headers });
    }

    if (url.pathname.startsWith("/api/admin/")) {
      if (!adminAuthorized) return Response.json({ error: "Unauthorized" }, { status: 401 });
      if (!sameOrigin(request)) return Response.json({ error: "Invalid origin" }, { status: 403 });
      if (url.pathname === "/api/admin/dashboard-data") return dataAdmin(request, env);
      if (url.pathname === "/api/admin/users" && request.method === "GET") {
        const result = await env.DB.prepare("SELECT id,name,username,can_amos,can_aog,can_fak,can_marshaling,can_permits,can_export,is_admin,active,created_at,updated_at FROM employees ORDER BY name COLLATE NOCASE").all();
        return Response.json({ users: result.results }, { headers: { "Cache-Control": "no-store" } });
      }
      if (url.pathname === "/api/admin/users" && request.method === "POST") {
        const body = await request.json<Record<string, unknown>>();
        const name = String(body.name || "").trim(), username = String(body.username || "").trim().toLowerCase(), pass = String(body.password || "");
        if (!name || !/^[a-z0-9._-]{3,40}$/.test(username) || pass.length < 8) return Response.json({ error: "Enter a name, a valid username, and a password of at least 8 characters." }, { status: 400 });
        const salt = randomHex(), hash = await passwordHash(pass, salt), now = new Date().toISOString();
        try {
          await env.DB.prepare("INSERT INTO employees (name,username,password_hash,password_salt,can_amos,can_aog,can_fak,can_marshaling,can_permits,can_export,is_admin,active,created_at,updated_at) VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?)")
            .bind(name, username, hash, salt, body.canAmos ? 1 : 0, body.canAog ? 1 : 0, body.canFak ? 1 : 0, body.canMarshaling ? 1 : 0, body.canPermits ? 1 : 0, body.canExport ? 1 : 0, body.isAdmin === true ? 1 : 0, 1, now, now).run();
          return Response.json({ ok: true });
        } catch { return Response.json({ error: "That username is already in use." }, { status: 409 }); }
      }
      const match = url.pathname.match(/^\/api\/admin\/users\/(\d+)$/);
      if (match && request.method === "PATCH") {
        const body = await request.json<Record<string, unknown>>(), id = Number(match[1]), now = new Date().toISOString();
        await env.DB.prepare("UPDATE employees SET name=?,can_amos=?,can_aog=?,can_fak=?,can_marshaling=?,can_permits=?,can_export=?,is_admin=?,active=?,updated_at=? WHERE id=?")
          .bind(String(body.name || "").trim(), body.canAmos ? 1 : 0, body.canAog ? 1 : 0, body.canFak ? 1 : 0, body.canMarshaling ? 1 : 0, body.canPermits ? 1 : 0, body.canExport ? 1 : 0, body.isAdmin === true ? 1 : 0, body.active ? 1 : 0, now, id).run();
        const newPassword = String(body.password || "");
        if (newPassword) { if (newPassword.length < 8) return Response.json({ error: "Password must contain at least 8 characters." }, { status: 400 }); const salt = randomHex(), hash = await passwordHash(newPassword, salt); await env.DB.prepare("UPDATE employees SET password_hash=?,password_salt=?,updated_at=? WHERE id=?").bind(hash, salt, now, id).run(); }
        return Response.json({ ok: true });
      }
      if (match && request.method === "DELETE") { await env.DB.prepare("DELETE FROM employees WHERE id=?").bind(Number(match[1])).run(); return Response.json({ ok: true }); }
      return Response.json({ error: "Not found" }, { status: 404 });
    }

    if (url.pathname === "/dashboard-source.html" || url.pathname === "/dashboard-source") {
      return new Response("Not found", { status: 404, headers: { "Cache-Control": "no-store" } });
    }

    if (url.pathname === "/portal" || url.pathname === "/dashboard" || url.pathname === "/secure-dashboard" || url.pathname === "/admin") {
      if (!authorized && !(url.pathname === "/secure-dashboard" && grant && url.searchParams.get("tab") === grant.dashboard)) return Response.redirect(new URL("/login", request.url), 303);
    }

    if (url.pathname === "/admin" && !adminAuthorized) return Response.redirect(new URL("/login?admin=1", request.url), 303);

    if (url.pathname === "/dashboard" || url.pathname === "/secure-dashboard") {
      const requestedTab = url.searchParams.get("tab") || "amos";
      if (employee && (!dashboardKeys.includes(requestedTab as typeof dashboardKeys[number]) || !userPermissions[requestedTab as keyof typeof userPermissions])) return Response.redirect(new URL("/portal?denied=1", request.url), 303);
      if (requestedTab === "permits" && !adminAuthorized && !employee?.can_permits) return Response.redirect(new URL("/portal?denied=1", request.url), 303);
    }

    if (url.pathname === "/secure-dashboard") {
      if (url.searchParams.get("tab") === "permits") {
        let html = permitsDashboardHtmlSource;
        if (url.searchParams.get("embedded") === "1") {
          html = html.replace("</head>", `<style>.topbar{display:none!important}.main{padding-top:13px!important}body{overflow-x:hidden!important}</style></head>`);
        }
        return new Response(html, { headers: {
          "Content-Type": "text/html; charset=utf-8",
          "Cache-Control": "private, no-store",
          "X-Frame-Options": "SAMEORIGIN",
          "X-Content-Type-Options": "nosniff",
        }});
      }
      let html = dashboardHtmlSource;
      const requestedTab = url.searchParams.get("tab") || "";
      const tab = ["amos", "aog", "fak", "marshaling"].includes(requestedTab) ? requestedTab : "amos";
      if (url.searchParams.get("embedded") === "1") {
        html = html.replace("</head>", `<style>.topbar{display:none!important}body{overflow-x:hidden!important}.frame-area{width:100%!important}</style></head>`);
      }
      html = html.replace(/<\/body>\s*<\/html>\s*$/, `<script>try{showTab("${tab}")}catch(e){}</script></body></html>`);
      return new Response(html, { headers: {
        "Content-Type": "text/html; charset=utf-8",
        "Cache-Control": "private, no-store",
        "X-Frame-Options": "SAMEORIGIN",
        "X-Content-Type-Options": "nosniff",
      }});
    }

    if (url.pathname === "/_vinext/image") {
      const allowedWidths = [...DEFAULT_DEVICE_SIZES, ...DEFAULT_IMAGE_SIZES];
      return handleImageOptimization(request, {
        fetchAsset: (path) => env.ASSETS.fetch(new Request(new URL(path, request.url))),
        transformImage: async (body, { width, format, quality }) => {
          const result = await env.IMAGES.input(body).transform(width > 0 ? { width } : {}).output({ format, quality });
          return result.response();
        },
      }, allowedWidths);
    }

    const routedHeaders = new Headers(request.headers);
    routedHeaders.set("x-sal-role", adminAuthorized ? "admin" : employee ? "employee" : generalAuthorized ? "general" : "anonymous");
    routedHeaders.set("x-sal-name", employee?.name || (adminAuthorized ? "Administrator" : "Guest"));
    routedHeaders.set("x-sal-permissions", JSON.stringify(userPermissions));
    routedHeaders.set("x-sal-can-export", employee?.can_export ? "1" : adminAuthorized ? "1" : "0");
    return handler.fetch(new Request(request, { headers: routedHeaders }), env, ctx);
  },
};

export default worker;
